    \begin{definisjon}
        Et $m$-kryss er en union av $m$ distinkte linjer $l_1, l_2, \dots, l_m$ i $\mathbb{P}^3$ som skjærer hverandre i et felles punkt $p$ og ligger i et felles plan $H$. Vi skriver $Z_m = \bigcup_{i=1}^m l_i$.
    \end{definisjon}
\end{frame}

\begin{frame}{$m$-kryss}
    \begin{teorem}
        Gitt et $m$-kryss, kan vi finne en glatt flate $S_d$ av grad $d$ som inneholder det hvis og bare hvis $d = 1$ eller $d \geq m$.
    \end{teorem}

    Nøkkelidé: Automorfi av $\mathbb{P}^3$ gir
    % can be used to move an $m$-cross into a position where we can construct such a surface. Explicitly, we move the lines to 
    \[l_1 = (\alpha : \beta : 0 : 0), \ l_2 = (\alpha : 0 : \gamma : 0), \ l_3=(\alpha: \lambda_3 \delta : \delta: 0), \dots, l_m = (\alpha :\lambda_m  \delta : \delta : 0)\]

    Disse linjene skjærer alle i punktet $p = (1 : 0 : 0 : 0)$ og ligger i planet $H = V(x_3)$.
\end{frame}

\begin{frame}{$m$-kryss}
    \begin{itemize}
        \item For $d=1$ så kan vi ta $X = H$.
        \item For $1 < d < m$ finnes det ingen slik flate ifølge Bezouts teorem. 
        % Consider a surface $X \subset \mathbb{P}^3$ of degree $d$ containing an $m$-cross $Z_m$, and a line in $H$ not passing through $p$. We use Bezout's theorem to see that this line must intersect $X$ in at least $d$ points, and since it intersects each line, we must have $d \geq m$ when $X$ is not equal to $H$.
        \item For $d \geq m$ kan vi bevise at det finnes en slik flate. 
        % We consider a linear system with base locus $Z_m$, and find a surface which is smooth on $Z_m$. Then we apply Bertini's theorem to see that a general member of the linear system is smooth away from $Z_m$. Then we can find a surface of degree $m$ containing $Z_m$.
    \end{itemize}

    Det lineære systemet
    \[\mathfrak{L} = \{X \mid Z_m \subseteq X\}\]
    har baselokus $Z_m$. Vi leter etter en flate i $\mathfrak{L}$ som er glatt på $Z_m$.

    Linjene beskrives av $l_i = V(x_3, g_i)$, vi får
    \[l_1 = V(x_3, x_2), \ l_2 = V(x_3, x_1), \ l_3 = V(x_3, x_1 - \lambda_3 x_2), \ \dots, \ l_m = V(x_3, x_1 - \lambda_m x_2).\]

    Definer funksjonen $f = \prod g_i + x_0^{m-1}x_3$.
    \[f(x_0, x_1, x_2, x_3) = x_1x_2 \prod_{i=3}^m (x_1 - \lambda_i x_2) + x_0^{m-1}x_3.\]

    Kan vise at $f$ er glatt på $Z_m$, og anvende Bertinis teorem for å få en glatt flate i $\mathfrak{L}$.
\end{frame}

\begin{frame}{$m$-kryss}
    \begin{teorem}
        La $A_m = \{X \subseteq \mathbb{P}^3 \mid Z_m \subset X\}$ være mengden av grad $d$ flater som inneholder et generelt $Z_m$. Da er $\dim{A_m} = \binom{d-m+2}{2} + \binom{d+2}{3} - 1$.
    \end{teorem}
\end{frame}

\begin{frame}[fragile]{$m$-sykel}
    \begin{definisjon}
        En $m$-sykel er en union av $m$ linjer $l_1, l_2, \dots, l_m$ slik at $l_i$ skjærer $l_{i+1}$ for $1 \leq i < m$, og $l_m$ skjærer $l_1$. Vi skriver $C_m = \bigcup_{i=1}^m l_i$ og betegner skjæringspunktene som $p_i = l_i \cap l_{i+1}$ for $1 \leq i < m$, og $p_m = l_m \cap l_1$.
    \end{definisjon}
    Vi sier at en $m$-sykel er \emph{regulær} hvis ingen linjer utenom de krevde skjærer, og det ikke finnes doble linjer. 
    
\end{frame}

\begin{frame}[fragile]{$m$-sykel}
        \begin{teorem}
        Gitt en regulær $m$-sykel, kan vi finne en flate $X$ av grad $d$ som inneholder den når $m < \binom{d+3}{3}\cdot \frac{1}{d}$.
    \end{teorem}
    
    Vi ser på den eksakte sekvensen
    \[0 \to \mathcal{I}_{C_m}(d) \to \mathcal{O}_{\mathbb{P}^3}(d) \to \mathcal{O}_{C_m}(d) \to 0.\]
    Tar vi kohomologi, får vi den lange eksakte sekvensen
    \[
        \begin{tikzcd}[sep=small]
        0 \ar[r]
        & H^0(\mathcal{I}_{C_m}(d)) \ar[r]
        & H^0(\mathcal{O}_{\mathbb{P}^3}(d)) \ar[r]
        & H^0(\mathcal{O}_{C_m}(d)) \ar[controls={+(3,-1.5) and +(-3,1.5)}]{dll} \\
        {}
        & H^1(\mathcal{I}_{C_m}(d)) \ar[r]
        & H^1(\mathcal{O}_{\mathbb{P}^3}(d)) \ar[r]
        & H^1(\mathcal{O}_{C_m}(d)) \ar[controls={+(3,-1.5) and +(-3,1.5)}]{dll} \\
        {}
        & H^2(\mathcal{I}_{C_m}(d)) \ar[r]
        & H^2(\mathcal{O}_{\mathbb{P}^3}(d)) \ar[r]
        & \dots
        \end{tikzcd}
    \]
    Vil se når $h^0(\mathcal{I}_{C_m}(d)) > 0$.
\end{frame}

\begin{frame}{$m$-sykel}
    Når vi tar dimensjoner, så får vi
    \[h^0(\mathcal{I}_{C_m}(d)) = h^0(\mathcal{O}_{\mathbb{P}^3}(d)) + h^1(\mathcal{I}_{C_m}(d)) - h^0(\mathcal{O}_{C_m}(d)).\]
    Vi vet at
    \[h^0(\mathcal{O}_{\mathbb{P}^3}(d)) = \binom{d+3}{3} \text{ og } h^0(\mathcal{O}_{C_m}(d)) = md.\]
    Så
    \[h^0(\mathcal{I}_{C_m}(d)) = \binom{d+3}{3} - md + h^1(\mathcal{I}_{C_m}(d)).\]
\end{frame}

\begin{frame}{$m$-sykel}
    % $h^0(\mathcal{I}_{C_m}(d))$ tells us the dimension of the space of degree $d$ surfaces including the $m$-gon $C_m$. 
    Siden $h^1(\mathcal{I}_{C_m}(d)) \geq 0$, så har vi at
    \[h^0(\mathcal{I}_{C_m}(d)) \geq \binom{d+3}{3} - md,\]
    og vi kan derfor garantere en flate når $\binom{d+3}{3} - md > 0$, altså $m < \binom{d+3}{3} \cdot \frac{1}{d}$.
\end{frame}

\begin{frame}{$m$-kjede}
    \begin{definisjon}
        En $m$-kjede er en union av $m$ linjer $l_1, l_2, \dots, l_m$ slik at $l_i$ skjærer $l_{i+1}$ for $1 \leq i < m$. Vi skriver $Q_m = \bigcup_{i=1}^m l_i$ og betegner skjæringspunktene som $p_i = l_i \cap l_{i+1}$ for $1 \leq i < m$.

    En $m$-kjede er \emph{regulær} hvis den kun skjærer de forventede linjene, og det ikke finnes doble linjer.

\begin{frame}{6-kjede}
    % Recall that a divisor $D$ on a variety $S$ is \emph{ample} if some positive multiple of $D$ defines an embedding of $S$ into projective space. We now state the Hodge Index Theorem for surfaces.
    \begin{teorem}[Hodge Index]
        La $H$ være en ampel divisor på en glatt flate $S$, og anta at $D$ er en divisor, $D \not\equiv 0$, og $D.H = 0$. Da har vi at $D^2 < 0$.
    \end{teorem}
\end{frame}

\begin{frame}{6-kjede}
    \begin{teorem}
       I en glatt kubisk flate kan man finne høyst en regulær 5-kjede.
    \end{teorem}
    \begin{itemize}
        \item Kan finne en regulær 5-kjede.
        \item Betrakt divisoren $D = C_1 + C_2 - C_4 - C_5$.
        \item $D^2 = 0$.
        \item $H = f^*(\mathcal{O}_{\mathbb{P}^3}(1))$ er ampel, og $H.D = 0$.
        \item Ved Hodge-indeksteoremet så får vi at $D = 0$.
        \item Så $C_1 + C_2 = C_4 + C_5$.
        \item $C_1.C_6 + C_2.C_6 = C_4.C_6 + 1$
    \end{itemize}

    % Let $C_i \neq C_4$ be another line which intersects $C_5$. Then it must also intersect either $C_1$ or $C_2$ by the equality, and thus cannot be used to make a regular 6-chain.
\end{frame}

    \begin{itemize}
        \item $q$ er proper. Å ta dimensjonen til fibre av $q$ gir en øvre semi-kontinuerlig funksjon.

        \item Fiberet til $q$ er mengden av løsninger til lineære likninger, så det definerer et projektivt rom.
        \item Dette rommet er $\mathbb{P}^0$, så flaten er unik.
    \end{itemize}
