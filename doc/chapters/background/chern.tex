\subsection{The first Chern class of a line bundle}

(\cite{EH16}, 1.4.)
Let $\mathcal{L}$ be a line bundle on a variety $X$ and $\sigma$ be a rational section. Then on an open affine set $U$ of a covering of $X$, we can write $\sigma|_U = f_U / g_U$, and define $\text{Div}(\sigma)|_U = \text{Div}(f_U) - \text{Div}(g_U)$. This definition agrees on the intersections of the open affine sets, and thus defines a Cartier divisor on $X$. Moreover, if $\tau$ is another rational section of $\mathcal{L}$, then $\alpha = \sigma / \tau$ is a well-defined rational function(!?), so $\mathrm{Div}(\sigma)$ and $\mathrm{Div}(\tau)$ are rationally equivalent divisors: 
\[\mathrm{Div}(\sigma) - \mathrm{Div}(\tau) = \mathrm{Div}(\alpha) \equiv 0 \text{ mod } \mathrm{Rat}(X).\]

Thus, we can define the first Chern class of the line bundle $\mathcal{L}$ as
\[c_1(\mathcal{L}) = [\mathrm{Div}(\sigma)] \in A^1(X),\]
the rational equivalence class $\mathrm{Div}(\sigma)$  in the Chow group $A^1(X)$, for any nonzero rational section $\sigma$. This is well-defined, as it does not depend on the choice of rational section $\sigma$.

\begin{example}
    A consequence is that $c_1(\mathcal{O}_{\mathbb{P}^n}(d))$ is the class of any hypersurface of degree $d$.
\end{example}

\subsection{Chern classes of vector bundles}

We now extend the definition of Chern classes to vector bundles of higher rank. Let $\mathcal{E}$ be a vector bundle on an $n$-dimensional variety $X$, and let $r = \mathrm{rank}(\mathcal{E})$. We follow the approach of \cite{EH16} section 5.2, giving an informal characterization rather than a formal definition.

In the case $r=1$, the Chern class $c_1(\mathcal{L})$ can be viewed as a measure of non-triviality. If $c_1(\mathcal{L}) = 0$, then $\mathcal{L}$ has a nowhere-vanishing section, and $\mathcal{L} \cong \mathcal{O}_X$. We extend this idea to higher rank vector bundles by considering the ``degeneracy locus'' of collections of sections. Roughly, this is the locus where the sections become linearly dependent in the fibers of $\mathcal{E}$.

More precisely, the bundle $\mathcal{E}$ is said to be trivial if and only if it has $r$ global sections  which are everywhere linearly independent. In this case, any set of $r$ general sections will do. \todo{Necessary?} We look at the exterior powers of $\mathcal{E}$. The $(r-i)$-th exterior power $\bigwedge^k \mathcal{E}$ is a vector bundle of rank $\binom{r}{r-i}$, and general sections $s_1, \dots, s_{r-i}$ induce sections of $\bigwedge^{r-i} \mathcal{E}$ given by $s_{1} \wedge s_{2} \wedge \dots \wedge s_{r-i}$. For $i=0$, we have a single section $s_1 \wedge s_2 \wedge \dots \wedge s_r$ of the line bundle $\bigwedge^r \mathcal{E}$, and the zero locus is by definition $c_1(\bigwedge^r \mathcal{E})$. This is a class in $A^1(X)$, and we call it the \emph{first Chern class of $\mathcal{E}$}.

In general, we have that for $(r-i)$ general sections, we can consider the vanishing locus of 
\[s_1 \wedge \dots \wedge s_{r-i} \in \bigwedge^{r-i} \mathcal{E}.\]
This vanishing locus is called the \emph{$i$-th degeneracy locus} of the sections $s_1, \dots, s_{r-i}$.

\subsection{Important results on Chern classes}

\begin{theorem}[\cite{EH16}, Theorem 5.3 (a), (c)]
    There is a unique way of assigning to each vector bundle $\mathcal{E}$ on a smooth quasi-projective variety $X$ a class $c(\mathcal{E}) = 1 + c_1(\mathcal{E}) + \ldots + c_n(\mathcal{E}) \in A(X)$ in such a way that
    \begin{itemize}
        \item If $\mathcal{L}$ is a line bundle, then $c(\mathcal{L})$ is $1 + c_1(\mathcal{L})$, where $c_1(\mathcal{L}) \in A^1(X)$ is the class of the divisor of zeros minus the divisor of poles of any rational section of $\mathcal{L}$.
        \item If $0 \to \mathcal{E} \to \mathcal{F} \to \mathcal{G} \to 0$ is an exact sequence of vector bundles on $X$, then $c(\mathcal{E}) = c(\mathcal{F}) c(\mathcal{G}) \in A(X)$.
    \end{itemize}
\end{theorem}

\begin{theorem}[\cite{EH16}, Theorem 5.12, Splitting principle]
    Any identity among Chern classes of bundles that is true for bundles that are direct sums of line bundles is true in general.
\end{theorem}

\begin{corollary}[\cite{EH16}, Corollary 5.5]
    If $\mathcal{E}$ is the direct sum of line bundles $\mathcal{L}_i$, then 
    \[c(\mathcal{E}) = \prod c(\mathcal{L}_i) = \prod (1 + c_1(\mathcal{L}_i)).\]
\end{corollary}

\begin{corollary}[\cite{EH16}, Example 5.13]
    If $\mathcal{E}$ is a rank $r$ vector bundle, then $c_i(\mathcal{E}) = 0$ for $i > r$.
